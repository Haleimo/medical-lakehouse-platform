# Python's stdlib `distutils` was removed in Python 3.12 (PEP 632). PySpark
# 3.5.x's pandas<->Spark conversion path (pyspark/sql/pandas/utils.py) still
# does `from distutils.version import LooseVersion` at import time, so on
# Python 3.12+ any code path that touches pandas conversion (e.g. the Gold
# aggregation step, which uses pandas under the hood) fails with
# `ModuleNotFoundError: No module named 'distutils'` -- confirmed by actually
# running the test suite on Python 3.12 here. Pinning setuptools below 81
# keeps its bundled `setuptools._distutils` shim active, which restores a
# working top-level `import distutils` without needing to touch pyspark or
# Python itself. Safe to remove once this project moves to pyspark>=4.1.
setuptools<81

# (org.apache.iceberg:iceberg-spark-runtime-*, org.projectnessie:*) are NOT
# pinned here -- they are supplied at spark-submit time via

pyspark>=3.5,<4.0
